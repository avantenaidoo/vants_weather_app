import { useState, useEffect } from 'react';
import { getCurrentWeather, get3DayForecast, get3DayHistory } from '../services/weatherService';
import { WeatherData } from '../types/weather'; 

type UseWeatherReturn = {
  currentWeather: WeatherData | null;
  forecast: WeatherData[];
  history: WeatherData[];
  loading: boolean;
  error: string | null;
  fetchWeather: (city: string) => void;
};

const useWeather = (): UseWeatherReturn => {
  const [currentWeather, setCurrentWeather] = useState<WeatherData | null>(null);
  const [forecast, setForecast] = useState<WeatherData[]>([]);
  const [history, setHistory] = useState<WeatherData[]>([]);
  const [loading, setLoading] = useState<boolean>(false);
  const [error, setError] = useState<string | null>(null);

  const fetchWeather = async (city: string) => {
    setLoading(true);
    setError(null);

    try {
      // Open IndexedDB
      const db = await openIndexedDB('weatherDB', 1);

      const cachedData = await getFromIndexedDB(db, city);

      if (cachedData) {
        setCurrentWeather(cachedData.currentWeather);
        setForecast(cachedData.forecast);
        setHistory(cachedData.history);
      } else {
        const current = await getCurrentWeather(city);
        const forecastData = await get3DayForecast(city);
        const historyData = await get3DayHistory(city);

        const weatherData = {
          currentWeather: {
            location: {
              name: city,
            },
            current: {
              date: current.observation_time,
              temperature: current.temperature,
              description: current.weather_descriptions[0],
              humidity: current.humidity,
              wind_speed: current.wind_speed,
              icon: current.weather_icons[0],
            },
          },
          forecast: forecastData.map((day: any) => ({
            location: {
              name: city,
            },
            current: {
              date: day.date,
              temperature: day.temperature,
              description: day.weather_descriptions[0],
              humidity: day.humidity,
              wind_speed: day.wind_speed,
              icon: day.weather_icons[0],
            },
          })),
          history: historyData.map((day: any) => ({
            location: {
              name: city,
            },
            current: {
              date: day.date,
              temperature: day.temperature,
              description: day.weather_descriptions[0],
              humidity: day.humidity,
              wind_speed: day.wind_speed,
              icon: day.weather_icons[0],
            },
          })),
        };

        // Store in IndexedDB
        await saveToIndexedDB(db, city, weatherData);

        setCurrentWeather(weatherData.currentWeather);
        setForecast(weatherData.forecast);
        setHistory(weatherData.history);
      }
    } catch (err) {
      console.error('Failed to fetch weather data:', err);
      setError('Failed to fetch weather data');
    } finally {
      setLoading(false);
    }
  };

  const openIndexedDB = (dbName: string, version: number): Promise<IDBDatabase> => {
    return new Promise((resolve, reject) => {
      const request = indexedDB.open(dbName, version);

      request.onupgradeneeded = (event) => {
        const db = (event.target as IDBRequest).result;
        if (!db.objectStoreNames.contains('weather')) {
          db.createObjectStore('weather', { keyPath: 'city' });
        }
      };

      request.onsuccess = (event) => {
        resolve((event.target as IDBRequest).result);
      };

      request.onerror = () => {
        reject('Failed to open IndexedDB');
      };
    });
  };

  const getFromIndexedDB = (db: IDBDatabase, city: string): Promise<{ currentWeather: WeatherData; forecast: WeatherData[]; history: WeatherData[] } | null> => {
    return new Promise((resolve, reject) => {
      const transaction = db.transaction('weather', 'readonly');
      const store = transaction.objectStore('weather');
      const request = store.get(city);

      request.onsuccess = () => {
        resolve(request.result ? request.result : null);
      };

      request.onerror = () => {
        reject('Failed to get data from IndexedDB');
      };
    });
  };

  const saveToIndexedDB = (db: IDBDatabase, city: string, weatherData: { currentWeather: WeatherData; forecast: WeatherData[]; history: WeatherData[] }): Promise<void> => {
    return new Promise((resolve, reject) => {
      const transaction = db.transaction('weather', 'readwrite');
      const store = transaction.objectStore('weather');
      const request = store.put({ city, ...weatherData });

      request.onsuccess = () => {
        resolve();
      };

      request.onerror = () => {
        reject('Failed to save data to IndexedDB');
      };
    });
  };

  return {
    currentWeather,
    forecast,
    history,
    loading,
    error,
    fetchWeather,
  };
};

export default useWeather;
